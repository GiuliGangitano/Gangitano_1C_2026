328void Calculo_Pitch_Roll(float *p, float *r)
329{
330 MPU6050_getAcceleration(&acc_x, &acc_y, &acc_z);
331 accf_x = (float_t)acc_x;
332 accf_y = (float_t)acc_y;
333 accf_z = (float_t)acc_z;
334 *p = atan2f(-accf_x, sqrtf((accf_y * accf_y) + (accf_z * accf_z))) * 180 / (float_t)M_PI;
335 *r = atan2f(accf_y, accf_z) * 180 / (float_t)M_PI;
336}

482void Recibir_Comando_UART(void *param)
483{
484 uint8_t caracter;
485 memset(buffer_entrada, 0, sizeof(buffer_entrada));
486
487 UartReadBuffer(UART_PC, &buffer_entrada[0], MAX_DIGITOS + 1);
488 caracter = buffer_entrada[0];
489
490 if (config_rut != CONFIG_NINGUNO)
491 {
492 Configuracion_Rutina_UART(caracter, (char *)&buffer_entrada[0]);
493 return;
494 }
495 if (caracter == 's' || caracter == 'S')
496 {
497 if (estado_sist == ESTADO_STANDBY)
498 {
499 if (!referencia_registrada)
500 {
501 printf("\nPresione TECLA 1 para setear referencia\n");
502 }
503 else if (!rutina_configurada)
504 {
505 printf("\nConfigure la rutina primero.\n");
506 Iniciar_Configuracion_Rutina();
507 }
508 else
509 {
510 estado_sist = ESTADO_EJERCICIO;
511 printf("\nMedición INICIADA.\n");
512 xTaskNotifyGive(Rutina_task_handle);
513 }
514 }
515 else if (estado_sist == ESTADO_EJERCICIO || estado_sist == ESTADO_DESCANSO)
516 {
517 estado_sist = ESTADO_STANDBY;
518 rutina_configurada = false;
519 printf("\nMedición DETENIDA.\n");
520 }
521 else
522 {
523 printf("\nComando no disponible en el estado actual del sistema.\n");
524 }
525 }
526 else
527 {
528 printf("\nTecla '%c' no reconocida. Presione 'S' para iniciar/detener la medición.\n", caracter);
529 }

577 LedOff(LED_2);

585 BuzzerOff();

587 case ESTADO_EJERCICIO:

589 LedOff(LED_2);
590 LedOff(LED_3);
591 if ((pitch > pitch_ref + tolerancia) || (pitch < pitch_ref - tolerancia) ||
592 (roll > roll_ref + tolerancia) || (roll < roll_ref - tolerancia))
593 {
594 BuzzerSetFrec(NOTE_C6);
595 BuzzerOn();
596 }
597 else
598 {
599 BuzzerOff();
600 }
601 break;
602 case ESTADO_DESCANSO:
603 LedOn(LED_1);
604 LedOn(LED_2);
605 LedOff(LED_3);
606 BuzzerOff();
607 break;
608 }

684static void vTask_Deteccion_Pitch_Roll(void *pvParameter)
685{
686 bool mensaje_inicio_mostrado = false;

689 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);

705 pitch_ref, roll_ref);
706 flag_set_referencia = false;
707 referencia_registrada = true;
708 Iniciar_Configuracion_Rutina();
709 }
710 break;
711 case ESTADO_EJERCICIO:
712 Calculo_Pitch_Roll(&pitch, &roll);
713 printf("ax=%d ay=%d az=%d | Pitch_ref=%.2f Roll_ref=%.2f Pitch=%.2f Roll=%.2f\n",
714 acc_x, acc_y, acc_z, pitch_ref, roll_ref, pitch, roll);
715 Aplicar_Feedback();
716 break;
717 case ESTADO_DESCANSO:
718 Aplicar_Feedback();
719 break;

747static void vTask_Rutina(void *pvParameter)
748{
749 while (1)
750 {
751 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
752 rutina_interrumpida = false;
753 Buzzer_Inicio_Fin(); // Tono inicio de rutina
754 for (uint8_t serie_actual = 1; serie_actual <= series && !rutina_interrumpida; serie_actual++){
755 if(serie_actual > 1){
756 Buzzer_Transicion(); // Tono a partir de finalizar el primer descanso y comenzar la segunda serie
757 }
758 estado_sist = ESTADO_EJERCICIO;
759 printf("\n=== Serie %d/%d | EJERCICIO: %ds ===\n", serie_actual, series, ejercicio);
760 for (uint16_t t = ejercicio; t > 0; t--){
761 if (estado_sist != ESTADO_EJERCICIO)
762 {
763 rutina_interrumpida = true;
764 break;
765 }
766 vTaskDelay(1000/portTICK_PERIOD_MS);
767 }
768 if (rutina_interrumpida){
769 break;
770 }
771 if (serie_actual < series && descanso > 0){
772 Buzzer_Transicion(); // Tono inicio de descanso
773 estado_sist = ESTADO_DESCANSO;
774 printf("\n=== Serie %d/%d | DESCANSO: %ds ===\n", serie_actual, series, descanso);
775 for (uint16_t t = descanso; t > 0; t--)
776 {
777 if (estado_sist != ESTADO_DESCANSO)
778 {
779 rutina_interrumpida = true;
780 break;
781 }
782 vTaskDelay(1000/portTICK_PERIOD_MS);
783 }
784 }
785 if (rutina_interrumpida){
786 break;
787 }
788 }
789 if (!rutina_interrumpida)
790 {
791 printf("\n¡Rutina completada! %d series realizadas.\n", series);
792 }
793 else
794 {
795 printf("\nRutina interrumpida por el usuario.\n");
796 }
797 Buzzer_Inicio_Fin(); // Tono fin de rutina
798 estado_sist = ESTADO_STANDBY;
799 rutina_configurada = false;
800 }

841 .func_p = Atender_timer_MPU6050,
842 .param_p = NULL};
843 TimerInit(&timer_MPU6050);
844 TimerStart(timer_MPU6050.timer);
